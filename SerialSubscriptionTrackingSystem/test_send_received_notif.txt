$testEmails = [
    'tpu' => 'fobov47923@muncloud.com',
    'gsps' => 'jokepa9809@muncloud.com',
    'inspection' => 'rayipi9335@exespay.com',
    'admin' => 'sibohov673@exespay.com',
    'supplier' => 'febagev799@fabaos.com',
];

echo "╔════════════════════════════════════════════════════════════╗\n";
echo "║          RECEIVED NOTIFICATION EMAIL TEST                 ║\n";
echo "╚════════════════════════════════════════════════════════════╝\n\n";

echo "📧 Updating test user email addresses...\n";
foreach ($testEmails as $role => $email) {
    $user = \App\Models\User::where('role', $role)->first();
    if ($user) {
        try {
            $user->update(['email' => $email]);
            echo "   ✓ {$role}: {$email}\n";
        } catch (\Exception $e) {
            echo "   ℹ {$role}: Email update skipped (may already exist)\n";
        }
    }
}

echo "\n🔍 Finding subscription with any serial for testing...\n";
$subscription = \App\Models\Subscription::where('serial_title', 'ABCD')->first() ?? \App\Models\Subscription::where('serial_title', '4 Seasons')->first();
echo "   ✓ Found: " . $subscription->serial_title . "\n";
echo "   • Supplier: " . $subscription->supplier_name . "\n";

// Get any issue, preferring 'for_delivery' but accepting others for testing
$issue = \App\Models\SerialIssue::where('subscription_id', $subscription->_id)->where('status', 'for_delivery')->first() 
    ?? \App\Models\SerialIssue::where('subscription_id', $subscription->_id)->first();

if (!$issue) {
    echo "   ✗ ERROR: No issues found for this subscription!\n";
    exit();
}

echo "   ✓ Issue #" . $issue->issue_number . " found\n";
echo "   • Current status: " . $issue->status . "\n";
echo "   • (For testing, sending 'received' notification regardless of current status)\n";

$beforeCount = \App\Models\UserNotification::count();
echo "\n📊 Notifications BEFORE: $beforeCount\n";

echo "\n🚀 Triggering RECEIVED notification...\n";
\App\Services\ProcessMovementService::createStatusNotifications(
    'received',
    $subscription->serial_title . " - Issue #" . $issue->issue_number,
    (string) $subscription->_id,
    "ISSUE-" . $issue->issue_number,
    $subscription->supplier_name,
    (string) $issue->_id,
    'gsps',
    'Test GSPS User'
);

echo "   ✓ Notification queued!\n";

$afterCount = \App\Models\UserNotification::count();
echo "\n📬 Notifications AFTER: $afterCount (+" . ($afterCount - $beforeCount) . " new)\n";

echo "\n📧 New notifications created:\n";
$newNotifications = \App\Models\UserNotification::orderBy('created_at', 'desc')->limit($afterCount - $beforeCount)->get();
foreach ($newNotifications as $n) {
    $user = \App\Models\User::findById($n->user_id);
    $email = $user?->email ?? 'unknown';
    echo "   • " . strtoupper($n->user_role) . " ({$email}): " . $n->title . "\n";
}

echo "\n⏳ Checking mail queue...\n";
$queuedJobs = \DB::table('jobs')->count();
echo "   Queued jobs: $queuedJobs\n";
